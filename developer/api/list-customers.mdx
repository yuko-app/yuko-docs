---
title: "List customers"
api: "GET /public/customers"
description: "Retrieves a paginated list of your store's customers, newest first."
---

Results are sorted by `created_at`, newest first. To fetch a single customer, use [Retrieve a customer](/developer/api/retrieve-a-customer).

<Info>Requires the `read:customers` scope. See [Scopes](/developer/api/scopes).</Info>

## Query parameters

<ParamField query="email" type="string">
  Return only the customer whose email address exactly matches this value.
</ParamField>

<ParamField query="platform" type="enum<string>">
  Return only customers synced from this platform. One of `shopify`, `woocommerce`.
</ParamField>

<ParamField query="per_page" type="integer" default={20}>
  The number of results per page. Values above `100` are treated as `100`.
</ParamField>

<ParamField query="page" type="integer" default={1}>
  The page number to retrieve. See [Classic pagination](/developer/api/classic-pagination).
</ParamField>

## Response

<ResponseField name="data" type="object[]">
  The customers on this page. Each customer has the properties of the [Customer object](/customer-identity-object).

  <Expandable title="customer properties">
    <ResponseField name="id" type="string<uuid>">
      Yuko's unique identifier for the customer.
    </ResponseField>
    <ResponseField name="first_name" type="string | null">
      The customer's first name.
    </ResponseField>
    <ResponseField name="last_name" type="string | null">
      The customer's last name.
    </ResponseField>
    <ResponseField name="email" type="string | null">
      The customer's email address.
    </ResponseField>
    <ResponseField name="phone" type="string | null">
      The customer's phone number.
    </ResponseField>
    <ResponseField name="platform" type="string">
      The ecommerce platform the customer was synced from, for example `shopify`.
    </ResponseField>
    <ResponseField name="platform_customer_id" type="string | null">
      The customer's ID in your store. Pass this value as `customer_id` to other endpoints.
    </ResponseField>
    <ResponseField name="state" type="enum<string>">
      `member` for active loyalty members, or `disabled` if the customer is banned from the loyalty program.
    </ResponseField>
    <ResponseField name="date_of_birth" type="string<date> | null">
      The customer's birthday in `YYYY-MM-DD` format, if collected.
    </ResponseField>
    <ResponseField name="referral_code" type="string | null">
      The customer's referral code. `null` if the customer has no referral code yet.
    </ResponseField>
    <ResponseField name="referral_url" type="string | null">
      The customer's referral link, built from the referral landing page URL plus `?ref=<referral_code>`. `null` if there's no referral code or no landing page URL is set.
    </ResponseField>
    <ResponseField name="points" type="object">
      The customer's points totals.

      <Expandable title="points properties">
        <ResponseField name="balance" type="integer">
          Current redeemable points balance. Always `0` for customers who are banned or excluded from the loyalty program by a customer tag.
        </ResponseField>
        <ResponseField name="earned" type="integer">
          Total points ever earned.
        </ResponseField>
        <ResponseField name="redeemed" type="integer">
          Total points redeemed.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="created_at" type="string<date-time>">
      When the customer was created in Yuko (ISO 8601).
    </ResponseField>
    <ResponseField name="updated_at" type="string<date-time>">
      When the customer was last updated in Yuko (ISO 8601).
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="meta" type="object">
  <Expandable title="meta properties">
    <ResponseField name="pagination" type="object">
      Pagination details. See [Classic pagination](/developer/api/classic-pagination).

      <Expandable title="pagination properties">
        <ResponseField name="current_page" type="integer">The page returned.</ResponseField>
        <ResponseField name="per_page" type="integer">The number of records per page.</ResponseField>
        <ResponseField name="total" type="integer">The total number of records.</ResponseField>
        <ResponseField name="last_page" type="integer">The number of the last page.</ResponseField>
        <ResponseField name="has_more" type="boolean">`true` if there are more pages after this one.</ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

## Errors

| Status | Code | When |
| :----- | :--- | :--- |
| `401` | `missing_token`, `invalid_token` | The API key is missing, wrong or revoked. |
| `403` | `insufficient_scope` | The key doesn't have `read:customers`. |
| `429` | | The key exceeded the [rate limit](/developer/api/rate-limits). |

<RequestExample>

```bash cURL
curl --request GET \
  --url "https://api.yukoapp.com/api/v1/public/customers?per_page=20&page=1" \
  --header "Authorization: Bearer py_xxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxx"
```

```javascript JavaScript
const params = new URLSearchParams({ per_page: 20, page: 1 });

const response = await fetch(
  `https://api.yukoapp.com/api/v1/public/customers?${params}`,
  {
    method: "GET",
    headers: {
      Authorization: "Bearer py_xxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxx",
    },
  }
);

const data = await response.json();
```

```php PHP
$query = http_build_query(["per_page" => 20, "page" => 1]);

$ch = curl_init();
curl_setopt($ch, CURLOPT_URL, "https://api.yukoapp.com/api/v1/public/customers?{$query}");
curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
curl_setopt($ch, CURLOPT_HTTPHEADER, [
    "Authorization: Bearer py_xxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxx",
]);

$response = curl_exec($ch);
curl_close($ch);

$data = json_decode($response, true);
```

</RequestExample>
